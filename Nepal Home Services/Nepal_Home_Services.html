import pandas as pd
import matplotlib.pyplot as plt

df_price = pd.read_excel("Nepal_Home_Services_Dataset.xlsx", sheet_name="Pricing and Ecommerce")

tier_map = {
    "Full Online Cart and Payment": "Full Online Cart",
    "Online Shop with Product Categories": "Digital Storefront",
    "App Based Booking with Fixed Pricing": "Digital Storefront",
    "Listed Prices with Online Booking": "Digital Storefront",
    "Online Booking Pay After Service": "Pay After Service",
    "Booking Form or Phone Inquiry Only": "Offline Inquiry",
    "Contact Form or Phone Inquiry Only": "Offline Inquiry",
    "Quote on Inquiry Only": "Offline Inquiry",
    "Consultation on Inquiry Only": "Offline Inquiry",
    "Directory Listing Only": "Offline Inquiry",
    "Phone Call Booking Only": "Offline Inquiry",
}

df_price["Tier"] = df_price["E-Commerce Status"].map(tier_map).fillna("Other")

cross = pd.crosstab(df_price["Pricing Visibility"], df_price["Tier"])

cross.plot(kind="bar", stacked=True, figsize=(9, 5))

plt.title("Pricing vs E-Commerce Tier")
plt.xlabel("Pricing Visibility")
plt.ylabel("Count")
plt.xticks(rotation=30, ha="right")
plt.tight_layout()
plt.savefig("chart7.png", dpi=150)
plt.show()
